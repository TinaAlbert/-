cmake_minimum_required(VERSION 3.16)
project(graphics_algorithms_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(graphics-tests tests/graphics_tests.cpp)
    add_test(NAME graphics-regression COMMAND graphics-tests)
endif()
if(WIN32)
    find_package(SDL2 2.0.5 CONFIG REQUIRED)
    add_executable(graphics-lab src/main.cpp)
    target_link_libraries(graphics-lab PRIVATE SDL2::SDL2 comdlg32)
    if(MSVC)
        target_compile_options(graphics-lab PRIVATE /utf-8 /W4)
        target_compile_definitions(graphics-lab PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_custom_command(TARGET graphics-lab POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:graphics-lab>)
else()
    message(STATUS "Interactive app requires Windows; building algorithm tests only.")
endif()
